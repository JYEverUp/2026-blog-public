## 安装原生的redis

```plain
wget http://download.redis.io/releases/redis-5.0.14.tar.gz
2、解压并重命名为redis
tar -xvf redis-5.0.14.tar.gz && mv redis-5.0.14 redis
3、编译
make
4、修改配置文件                                                                                           配置文件在文件夹中，如不改配置 直接替换即可
vim redis.conf
\#所有ip访问
bind 0.0.0.0
\#同一ip对内访问
protected-mode no
requirepass 123456

5、启动
[root@iZwz92n7dzukubibde00xgZ redis]# ./src/redis-server /usr/local/src/redis/redis.conf
```

## 部署vue项目到docker

https://blog.csdn.net/qq_42058138/article/details/131131169

```plain
# server {
#     listen       80;

#         index  index.html index.htm ;

server {
listen 80;
server_name 106.12.121.241;
# 静态文件部署
root /usr/share/nginx/html;
index index.html index.htm ;
# API反向代理
location /api {
proxy_pass http://106.12.121.241:56000/;
proxy_set_header Host $host;
proxy_set_header X-Real-IP $remote_addr;
proxy_method POST;
}
# index.html文件
location / {
try_files $uri $uri/ /index.html;
error_page 405 =200 @405;  #405页面处理
}
# login.html文件
location /login {
try_files $uri $uri/ /login.html;
error_page 405 =200 @405;  #405页面处理
}
location /qrIndex {
try_files $uri $uri/ /qrIndex.html;
error_page 405 =200 @405;  #405页面处理
}
location /resetPwd {
try_files $uri $uri/ /resetPwd.html;
error_page 405 =200 @405;  #405页面处理
}
location /toggle {
try_files $uri $uri/ /toggle.html;
error_page 405 =200 @405;  #405页面处理
}
location @405 {
proxy_set_header Host $host;
proxy_set_header X-Real-IP $remote_addr;
proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
#ip为后端服务地址
proxy_pass http://106.12.121.241:56000$request_uri ;
}

}
```
